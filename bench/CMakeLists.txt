# Benchmark: libqyaml only
add_executable(bench-qyaml bench_parse.c)
target_link_libraries(bench-qyaml qyaml)
target_compile_definitions(bench-qyaml PRIVATE YAML_DECLARE_STATIC)
target_compile_options(bench-qyaml PRIVATE -O3 -march=native)

# Benchmark: comparison against reference libyaml (if built)
set(LIBYAML_ROOT "${CMAKE_SOURCE_DIR}/reference/libyaml")
set(LIBYAML_LIB "${LIBYAML_ROOT}/build/libyaml.a")
set(LIBYAML_INC "${LIBYAML_ROOT}/include")
message(STATUS "Looking for reference libyaml at: ${LIBYAML_LIB}")
if(EXISTS "${LIBYAML_LIB}")
    message(STATUS "Found reference libyaml - building comparison benchmark")
    add_executable(bench-libyaml bench_parse.c)
    target_include_directories(bench-libyaml PRIVATE "${LIBYAML_INC}")
    target_link_libraries(bench-libyaml "${LIBYAML_LIB}")
    target_compile_definitions(bench-libyaml PRIVATE YAML_DECLARE_STATIC)
    target_compile_options(bench-libyaml PRIVATE -O3 -march=native)
else()
    message(STATUS "Reference libyaml not found - skipping comparison benchmark")
endif()
